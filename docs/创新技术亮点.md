### 1.1 樱花飘落效果

40 片樱花花瓣，使用纯 CSS 模拟花瓣形状，配合旋转飘落动画。

```typescript
// YunYueBlog/components/Sakura.tsx
"use client";

import { useEffect, useState } from 'react';

interface Petal {
  id: number;
  x: number;
  y: number;
  size: number;
  rotation: number;
  speed: number;
  delay: number;
}

export default function Sakura() {
  const [petals, setPetals] = useState<Petal[]>([]);
  
  useEffect(() => {
    // 生成 40 片花瓣
    const newPetals: Petal[] = Array.from({ length: 40 }, (_, i) => ({
      id: i,
      x: Math.random() * 100,                    // 随机水平位置
      y: -10 - Math.random() * 20,               // 从顶部开始
      size: 8 + Math.random() * 12,              // 随机大小
      rotation: Math.random() * 360,             // 随机初始旋转
      speed: 1 + Math.random() * 2,              // 随机下落速度
      delay: Math.random() * 15,                 // 随机延迟
    }));
    setPetals(newPetals);
  }, []);
  
  return (
    <div className="fixed inset-0 pointer-events-none z-50 overflow-hidden">
      {petals.map((petal) => (
        <div
          key={petal.id}
          className="absolute"
          style={{
            left: `${petal.x}%`,
            top: `${petal.y}%`,
            width: `${petal.size}px`,
            height: `${petal.size}px`,
            animation: `sakura-fall ${15 / petal.speed}s linear ${petal.delay}s infinite`,
          }}
        >
          {/* 花瓣形状 - 使用 border-radius 模拟 */}
          <div
            className="w-full h-full"
            style={{
              background: `linear-gradient(135deg, #ffb7c5 0%, #ff69b4 100%)`,
              borderRadius: '100% 0 100% 0',  // 关键：花瓣形状
              transform: `rotate(${petal.rotation}deg)`,
              boxShadow: '0 0 6px rgba(255, 105, 180, 0.3)',
              animation: `sakura-rotate ${3 + Math.random() * 4}s ease-in-out infinite alternate`,
            }}
          />
        </div>
      ))}
      
      <style jsx>{`
        @keyframes sakura-fall {
          0% {
            transform: translateY(-10vh) translateX(0);
            opacity: 1;
          }
          25% {
            transform: translateY(25vh) translateX(15px);
          }
          50% {
            transform: translateY(50vh) translateX(-10px);
          }
          75% {
            transform: translateY(75vh) translateX(20px);
          }
          100% {
            transform: translateY(110vh) translateX(-5px);
            opacity: 0.3;
          }
        }
        
        @keyframes sakura-rotate {
          0% { transform: rotate(0deg) scale(1); }
          100% { transform: rotate(360deg) scale(0.8); }
        }
      `}</style>
    </div>
  );
}
```

**创新点：**
- 使用 `border-radius: 100% 0 100% 0` 创建花瓣形状，无需图片资源
- 双层动画：外层控制下落轨迹，内层控制花瓣旋转
- 随机参数（大小、速度、延迟）避免所有花瓣同步运动

### 1.2 萤火虫飞舞效果

50 只萤火虫，采用双层动画架构：外层漂浮 + 内层呼吸发光。

```typescript
// YunYueBlog/components/Fireflies.tsx
"use client";

import { useEffect, useState } from 'react';

interface Firefly {
  id: number;
  x: number;
  y: number;
  size: number;
  color: string;
  floatDuration: number;
  floatDelay: number;
  glowDuration: number;
  glowDelay: number;
  trajectory: number;  // 4 种随机轨迹
}

export default function Fireflies() {
  const [fireflies, setFireflies] = useState<Firefly[]>([]);
  
  useEffect(() => {
    // 萤火虫颜色
    const colors = [
      'rgba(255, 255, 150, 0.8)',  // 暖黄
      'rgba(150, 255, 150, 0.8)',  // 翠绿
      'rgba(200, 200, 255, 0.8)',  // 冷蓝
    ];
    
    // 生成 50 只萤火虫
    const newFireflies: Firefly[] = Array.from({ length: 50 }, (_, i) => ({
      id: i,
      x: Math.random() * 100,
      y: Math.random() * 100,
      size: 3 + Math.random() * 5,
      color: colors[Math.floor(Math.random() * colors.length)],
      floatDuration: 15 + Math.random() * 20,      // 漂浮周期 15-35 秒
      floatDelay: Math.random() * 10,
      glowDuration: 3 + Math.random() * 5,         // 呼吸周期 3-8 秒
      glowDelay: Math.random() * 5,
      trajectory: Math.floor(Math.random() * 4),   // 4 种轨迹
    }));
    setFireflies(newFireflies);
  }, []);
  
  // 获取漂浮动画
  const getFloatAnimation = (trajectory: number) => {
    switch (trajectory) {
      case 0: return 'float-1';
      case 1: return 'float-2';
      case 2: return 'float-3';
      case 3: return 'float-4';
      default: return 'float-1';
    }
  };
  
  return (
    <div className="fixed inset-0 pointer-events-none z-40 overflow-hidden"
         style={{ mixBlendMode: 'screen' }}>  {/* 关键：混合模式 */}
      {fireflies.map((firefly) => (
        <div
          key={firefly.id}
          className="absolute"
          style={{
            left: `${firefly.x}%`,
            top: `${firefly.y}%`,
            animation: `${getFloatAnimation(firefly.trajectory)} ${firefly.floatDuration}s ease-in-out ${firefly.floatDelay}s infinite`,
          }}
        >
          {/* 外层：发光光晕 */}
          <div
            className="absolute inset-0 rounded-full"
            style={{
              width: `${firefly.size * 4}px`,
              height: `${firefly.size * 4}px`,
              background: `radial-gradient(circle, ${firefly.color} 0%, transparent 70%)`,
              transform: 'translate(-50%, -50%)',
              animation: `glow ${firefly.glowDuration}s ease-in-out ${firefly.glowDelay}s infinite alternate`,
            }}
          />
          
          {/* 内层：核心光点 */}
          <div
            className="absolute rounded-full"
            style={{
              width: `${firefly.size}px`,
              height: `${firefly.size}px`,
              background: firefly.color,
              boxShadow: `0 0 ${firefly.size * 2}px ${firefly.color}`,
              transform: 'translate(-50%, -50%)',
              animation: `glow ${firefly.glowDuration}s ease-in-out ${firefly.glowDelay}s infinite alternate`,
            }}
          />
        </div>
      ))}
      
      <style jsx>{`
        /* 4 种不同的漂浮轨迹 */
        @keyframes float-1 {
          0%, 100% { transform: translate(0, 0); }
          25% { transform: translate(30px, -20px); }
          50% { transform: translate(-10px, 30px); }
          75% { transform: translate(20px, 10px); }
        }
        
        @keyframes float-2 {
          0%, 100% { transform: translate(0, 0); }
          25% { transform: translate(-25px, 15px); }
          50% { transform: translate(20px, -25px); }
          75% { transform: translate(-15px, -10px); }
        }
        
        @keyframes float-3 {
          0%, 100% { transform: translate(0, 0); }
          33% { transform: translate(15px, 25px); }
          66% { transform: translate(-20px, -15px); }
        }
        
        @keyframes float-4 {
          0%, 100% { transform: translate(0, 0); }
          20% { transform: translate(-10px, -30px); }
          40% { transform: translate(25px, 10px); }
          60% { transform: translate(-15px, 20px); }
          80% { transform: translate(10px, -15px); }
        }
        
        /* 发光呼吸动画 */
        @keyframes glow {
          0% { opacity: 0.3; transform: translate(-50%, -50%) scale(0.8); }
          100% { opacity: 1; transform: translate(-50%, -50%) scale(1.2); }
        }
      `}</style>
    </div>
  );
}
```

**创新点：**
- 双层动画架构：外层控制大范围漂浮，内层控制发光呼吸
- 4 种随机轨迹避免所有萤火虫同向飞行的违和感
- 使用 `mix-blend-mode: screen` 实现发光叠加效果
- 随机周期（15-35 秒漂浮，3-8 秒呼吸）创造自然感

## 2. 物理交互系统

### 2.1 圆形转盘导航

手机端摒弃传统汉堡菜单，设计可旋转的圆形转盘。

```typescript
// YunYueBlog/components/MobileRotaryMenu.tsx
"use client";

import { useState, useRef } from 'react';
import { motion, useMotionValue, useSpring, useTransform } from 'framer-motion';
import Link from 'next/link';

interface MenuItem {
  href: string;
  label: string;
  icon: string;
}

export default function MobileRotaryMenu({ 
  items, 
  onClose 
}: { 
  items: MenuItem[];
  onClose: () => void;
}) {
  const itemCount = items.length;
  const angleStep = 360 / itemCount;
  const radius = 120;  // 转盘半径
  
  // 物理旋转控制
  const rotation = useMotionValue(0);
  const springRotation = useSpring(rotation, {
    stiffness: 200,    // 弹簧刚度
    damping: 25,       // 阻尼系数
  });
  
  // 拖拽旋转
  const handleDrag = (event: any, info: any) => {
    const angle = Math.atan2(info.offset.y, info.offset.x) * (180 / Math.PI);
    rotation.set(angle);
  };
  
  // 计算反向旋转（保持文字正向）
  const inverseRotation = useTransform(springRotation, (r) => -r);
  
  return (
    <motion.div
      initial={{ opacity: 0, scale: 0.8 }}
      animate={{ opacity: 1, scale: 1 }}
      exit={{ opacity: 0, scale: 0.8 }}
      className="fixed inset-0 z-50 bg-black/50 backdrop-blur-sm 
                 flex items-center justify-center"
      onClick={onClose}
    >
      <motion.div
        onClick={(e) => e.stopPropagation()}
        className="relative w-80 h-80"
        style={{ rotate: springRotation }}
        drag
        dragConstraints={{ left: 0, right: 0, top: 0, bottom: 0 }}
        onDrag={handleDrag}
      >
        {/* 中心按钮 */}
        <motion.div 
          className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 
                     w-20 h-20 bg-white rounded-full shadow-xl 
                     flex items-center justify-center z-10"
          style={{ rotate: inverseRotation }}
        >
          <button 
            onClick={onClose}
            className="text-2xl transition-transform hover:rotate-90 hover:text-red-500"
          >
            ✕
          </button>
        </motion.div>
        
        {/* 导航项 */}
        {items.map((item, index) => {
          const angle = angleStep * index;
          const x = Math.cos((angle - 90) * Math.PI / 180) * radius;
          const y = Math.sin((angle - 90) * Math.PI / 180) * radius;
          
          return (
            <motion.div
              key={item.href}
              className="absolute top-1/2 left-1/2 
                         w-16 h-16 bg-white rounded-full shadow-lg 
                         flex flex-col items-center justify-center
                         hover:scale-110 transition-transform"
              style={{
                x: x - 32,
                y: y - 32,
                rotate: inverseRotation,  // 反向旋转保持文字正向
              }}
              whileHover={{ scale: 1.15 }}
              whileTap={{ scale: 0.95 }}
            >
              <Link href={item.href} onClick={onClose}>
                <span className="text-xl">{item.icon}</span>
                <span className="text-[10px] mt-1 block text-center">
                  {item.label}
                </span>
              </Link>
            </motion.div>
          );
        })}
      </motion.div>
    </motion.div>
  );
}
```

**创新点：**
- 使用 Framer Motion 的物理弹簧动画（`stiffness: 200, damping: 25`）
- 导航项使用 `inverseRotation` 反向旋转补偿，文字始终保持正向
- 关闭按钮旋转 90 度变红的交互细节

### 2.2 点击涟漪效果

使用 Canvas 实现物理衰减的涟漪扩散。

```typescript
// YunYueBlog/components/ClickEffect.tsx
"use client";

import { useEffect, useRef } from 'react';

interface Ripple {
  x: number;
  y: number;
  radius: number;
  maxRadius: number;
  opacity: number;
  speed: number;
  color: string;
}

export default function ClickEffect() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const ripplesRef = useRef<Ripple[]>([]);
  
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    
    const ctx = canvas.getContext('2d')!;
    let animationId: number;
    
    // 调整画布大小
    const resize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    };
    resize();
    window.addEventListener('resize', resize);
    
    // 点击事件
    const handleClick = (e: MouseEvent) => {
      // 创建多个涟漪
      for (let i = 0; i < 3; i++) {
        ripplesRef.current.push({
          x: e.clientX,
          y: e.clientY,
          radius: 0,
          maxRadius: 50 + i * 30,
          opacity: 0.8 - i * 0.2,
          speed: 2 + i * 0.5,
          color: `hsla(${240 + i * 20}, 80%, 70%, `,
        });
      }
    };
    
    // 动画循环
    const animate = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      ripplesRef.current = ripplesRef.current.filter((ripple) => {
        // 更新涟漪
        ripple.radius += ripple.speed;
        ripple.opacity *= 0.96;  // 物理衰减
        
        // 绘制涟漪
        ctx.beginPath();
        ctx.arc(ripple.x, ripple.y, ripple.radius, 0, Math.PI * 2);
        ctx.strokeStyle = `${ripple.color}${ripple.opacity})`;
        ctx.lineWidth = 2;
        ctx.shadowBlur = 10;
        ctx.shadowColor = `${ripple.color}0.5)`;
        ctx.stroke();
        
        // 返回 true 保留未消失的涟漪
        return ripple.opacity > 0.01;
      });
      
      animationId = requestAnimationFrame(animate);
    };
    
    window.addEventListener('click', handleClick);
    animate();
    
    return () => {
      window.removeEventListener('resize', resize);
      window.removeEventListener('click', handleClick);
      cancelAnimationFrame(animationId);
    };
  }, []);
  
  return (
    <canvas
      ref={canvasRef}
      className="fixed inset-0 pointer-events-none z-[9998]"
    />
  );
}
```

**创新点：**
- 每次点击生成 3 个涟漪，创造层次感
- 使用物理衰减（速度乘以 0.96，透明度线性递减）
- Canvas 发光模糊效果（`shadowBlur`）

## 3. 开屏动画系统

```typescript
// YunYueBlog/components/SplashScreen.tsx
"use client";

import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';

export default function SplashScreen() {
  const [isVisible, setIsVisible] = useState(false);
  const [progress, setProgress] = useState(0);
  
  useEffect(() => {
    // 检查是否已显示过（每个会话只显示一次）
    const hasShown = sessionStorage.getItem('splash-shown');
    if (hasShown) return;
    
    setIsVisible(true);
    
    // 模拟加载进度
    const interval = setInterval(() => {
      setProgress((prev) => {
        if (prev >= 100) {
          clearInterval(interval);
          setTimeout(() => {
            setIsVisible(false);
            sessionStorage.setItem('splash-shown', 'true');
          }, 500);
          return 100;
        }
        return prev + Math.random() * 15;
      });
    }, 100);
    
    return () => clearInterval(interval);
  }, []);
  
  return (
    <AnimatePresence>
      {isVisible && (
        <motion.div
          initial={{ opacity: 1 }}
          exit={{ 
            opacity: 0,
            scale: 1.1,
            filter: 'blur(10px)',
          }}
          transition={{ duration: 0.5, ease: 'easeInOut' }}
          className="fixed inset-0 z-[9999] bg-gradient-to-br 
                     from-indigo-500 via-purple-500 to-pink-500 
                     flex flex-col items-center justify-center"
        >
          {/* 头像光环 */}
          <motion.div
            className="relative w-32 h-32 mb-8"
            animate={{ rotate: 360 }}
            transition={{ duration: 8, repeat: Infinity, ease: 'linear' }}
          >
            <div className="absolute inset-0 rounded-full bg-gradient-to-r 
                           from-yellow-400 via-pink-500 to-purple-500 
                           animate-spin-slow" />
            <div className="absolute inset-2 rounded-full bg-white 
                           flex items-center justify-center">
              <span className="text-6xl">👤</span>
            </div>
          </motion.div>
          
          {/* 博客标题 */}
          <motion.h1
            initial={{ y: 20, opacity: 0 }}
            animate={{ y: 0, opacity: 1 }}
            transition={{ delay: 0.3 }}
            className="text-3xl font-bold text-white mb-4"
          >
            YunYue's Blog
          </motion.h1>
          
          {/* 进度条 */}
          <div className="w-64 h-2 bg-white/20 rounded-full overflow-hidden">
            <motion.div
              className="h-full bg-white rounded-full"
              initial={{ width: 0 }}
              animate={{ width: `${Math.min(progress, 100)}%` }}
              transition={{ duration: 0.3 }}
            />
          </div>
          
          {/* 加载文字 */}
          <motion.p
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            transition={{ delay: 0.5 }}
            className="text-white/80 text-sm mt-4"
          >
            {progress < 100 ? '正在加载...' : '欢迎光临！'}
          </motion.p>
        </motion.div>
      )}
    </AnimatePresence>
  );
}
```

**创新点：**
- 使用 `sessionStorage` 保证每次会话只显示一次
- 退出时带模糊缩放效果（`filter: 'blur(10px)'`）
- 头像光环持续旋转动画

## 5. 背景轮播系统

支持纯渐色和图片轮播两种模式。

```typescript
// YunYueBlog/components/BackgroundSlider.tsx
"use client";

import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';

interface BackgroundConfig {
  mode: 'gradient' | 'image';
  gradients?: string[];
  images?: string[];
  interval?: number;
}

export default function BackgroundSlider({ config }: { config: BackgroundConfig }) {
  const [currentIndex, setCurrentIndex] = useState(0);
  
  const { mode, gradients, images, interval = 5000 } = config;
  const items = mode === 'gradient' ? gradients! : images!;
  
  useEffect(() => {
    const timer = setInterval(() => {
      setCurrentIndex((prev) => (prev + 1) % items.length);
    }, interval);
    
    return () => clearInterval(timer);
  }, [items.length, interval]);
  
  return (
    <div className="fixed inset-0 z-0">
      <AnimatePresence mode="wait">
        <motion.div
          key={currentIndex}
          initial={{ opacity: 0, scale: 1.1 }}
          animate={{ opacity: 1, scale: 1 }}
          exit={{ opacity: 0, scale: 0.9 }}
          transition={{ duration: 1.5, ease: 'easeInOut' }}
          className="absolute inset-0"
          style={{
            background: mode === 'gradient' 
              ? gradients![currentIndex]
              : `url(${images![currentIndex]}) center/cover no-repeat`,
          }}
        />
      </AnimatePresence>
      
      {/* 渐变遮罩 */}
      <div className="absolute inset-0 bg-gradient-to-b 
                     from-transparent via-transparent to-white/80 
                     dark:to-slate-900/80" />
    </div>
  );
}
```

## 6. 主题系统

支持深色/浅色切换，通过 localStorage 持久化。

```typescript
// YunYueBlog/components/ThemeProvider.tsx
"use client";

import { createContext, useContext, useState, useEffect } from 'react';

type Theme = 'light' | 'dark';

interface ThemeContextType {
  theme: Theme;
  toggleTheme: () => void;
}

const ThemeContext = createContext<ThemeContextType | null>(null);

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [theme, setTheme] = useState<Theme>('dark');  // 默认深色
  
  // 初始化主题
  useEffect(() => {
    const saved = localStorage.getItem('theme') as Theme;
    if (saved) {
      setTheme(saved);
    } else if (window.matchMedia('(prefers-color-scheme: dark)').matches) {
      setTheme('dark');
    }
  }, []);
  
  // 应用主题
  useEffect(() => {
    document.documentElement.classList.toggle('dark', theme === 'dark');
    localStorage.setItem('theme', theme);
  }, [theme]);
  
  const toggleTheme = () => {
    setTheme((prev) => (prev === 'light' ? 'dark' : 'light'));
  };
  
  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider');
  }
  return context;
}
```
