```javascript
const REQUIRED_FIELDS = [
  { key: 'title', snippet: `\n  title: "YunYue's Blog",` },
  { key: 'description', snippet: `\n  description: "个人博客",` },
  { key: 'social', snippet: `\n  social: {\n    github: "https://github.com/...",\n  },` },

  { key: 'geminiConfig', snippet: `\n  geminiConfig: {\n    modelId: "gemini-2.5-flash-lite",\n    systemPrompt: "...",\n    maxOutputTokens: 150,\n    temperature: 0.85,\n  },` },
  // ... 更多字段
];

function checkAndFixConfig(filePath) {
  let content = fs.readFileSync(filePath, 'utf-8');
  
  for (const field of REQUIRED_FIELDS) {
    // 检查字段是否存在
    if (!content.includes(`${field.key}:`)) {
      console.log(`⚠️  缺失字段: ${field.key}，正在修补...`);
      
      // 在 siteConfig 对象末尾插入字段
      const lastBraceIndex = content.lastIndexOf('}');
      content = 
        content.slice(0, lastBraceIndex) + 
        field.snippet + 
        '\n' + 
        content.slice(lastBraceIndex);
    }
  }
  
  fs.writeFileSync(filePath, content, 'utf-8');
  console.log('✅ 配置文件检查完成');
}
```

```python
# my-blog-manager/cms_core/api/deploy.py
class DeployManager:
    """双轨部署管理器"""
    
    def __init__(self, blog_dir: str, ssh_key_path: str):
        self.blog_dir = blog_dir
        self.ssh_key_path = ssh_key_path
    
    async def deploy_line_a(self):
        """A 线：静态部署到 GitHub Pages"""
        # 构建静态文件
        result = await asyncio.create_subprocess_exec(
            "npm", "run", "deploy",
            cwd=self.blog_dir,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        await result.wait()
        
        if result.returncode == 0:
            return {"success": True, "message": "A 线部署成功"}
        return {"success": False, "message": "A 线部署失败"}
    
    async def deploy_line_b(self, repo_url: str, branch: str = "main"):
        """B 线：源码部署到 Vercel"""
        
        # 设置专属 SSH 密钥
        env = os.environ.copy()
        env["GIT_SSH_COMMAND"] = f"ssh -i {self.ssh_key_path}"
        
        # Git 操作
        commands = [
            ["git", "add", "."],
            ["git", "commit", "-m", f"deploy: 自动部署 {datetime.now().isoformat()}"],
            ["git", "push", repo_url, branch]
        ]
        
        for cmd in commands:
            result = await asyncio.create_subprocess_exec(
                *cmd,
                cwd=self.blog_dir,
                env=env,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            await result.wait()
            
            if result.returncode != 0:
                return {"success": False, "message": f"命令失败: {' '.join(cmd)}"}
        
        return {"success": True, "message": "B 线部署成功"}
```

```python
import webview
import uvicorn
import socket
from threading import Thread

class BlogManagerApp:
    """博客管理桌面客户端"""
    
    def __init__(self):
        self.window = None
        self.server = None
        self.port = self.find_free_port()

        """自动分配空闲端口"""
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.bind(('', 0))
            return s.getsockname()[1]

        """启动 FastAPI 后端"""

        uvicorn.run(app, host="127.0.0.1", port=self.port)
    
    def start_frontend(self):
        """启动 Next.js 前端"""
        import subprocess
        # 生产模式
        subprocess.Popen(
            ["node", ".next/standalone/server.js"],
            cwd="my-blog-manager"
        )
    
    def on_window_closed(self):
        """窗口关闭时清理资源"""
        if self.server:
            self.server.shutdown()
    
    def run(self):
        """启动应用"""
        # 1. 启动后端
        server_thread = Thread(target=self.start_server, daemon=True)
        server_thread.start()
        
        # 2. 启动前端

        # 3. 创建无边框窗口

            width=1200,

            resizable=True,
            frameless=True,  # 无边框

        )
```

```typescript
// YunYueBlog/app/api/music/route.ts
import { NextResponse } from 'next/server';

// 网易云音乐 API
const MUSIC_API = {
  song: 'https://music.163.com/api/song/detail',
  lyric: 'https://music.163.com/api/song/lyric',
  audio: 'https://music.163.com/song/media/outer/url',
};

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const ids = searchParams.get('ids')?.split(',') || [];
  
  if (ids.length === 0) {
    return NextResponse.json({ error: '缺少歌曲 ID' }, { status: 400 });
  }
  
  try {
    // 并行请求歌曲详情和歌词
    const [songsRes, lyricsRes] = await Promise.all([
      // 获取歌曲详情
      fetch(`${MUSIC_API.song}?ids=[${ids.join(',')}]`),
      // 获取歌词
      Promise.all(
        ids.map(id => 
          fetch(`${MUSIC_API.lyric}?id=${id}&lv=1&tv=-1`)
            .then(res => res.json())
        )
      )
    ]);
    
    const songsData = await songsRes.json();
    
    // 组装返回数据
    const songs = songsData.songs.map((song: any, index: number) => ({
      id: song.id,
      title: song.name,
      artist: song.artists.map((a: any) => a.name).join(', '),
      album: song.album.name,
      cover: song.album.picUrl + '?param=300y300',
      url: `${MUSIC_API.audio}?id=${song.id}.mp3`,
      lyric: parseLyric(lyricsRes[index]?.lrc?.lyric || ''),
    }));
    
    return NextResponse.json({ success: true, data: songs });
  } catch (error) {
    return NextResponse.json({ error: '获取歌曲失败' }, { status: 500 });
  }
}

// 歌词解析
function parseLyric(lrc: string): LyricLine[] {
  const lines = lrc.split('\n');
  const result: LyricLine[] = [];
  
  for (const line of lines) {
    const match = line.match(/\[(\d{2}):(\d{2})\.(\d{2,3})\](.*)/);
    if (match) {
      const minutes = parseInt(match[1]);
      const seconds = parseInt(match[2]);
      const milliseconds = parseInt(match[3]);
      const time = minutes * 60 + seconds + milliseconds / 1000;
      const text = match[4].trim();
      
      if (text) {
        result.push({ time, text });
      }
    }
  }
  
  return result.sort((a, b) => a.time - b.time);
}
```

### 2.2 全局音乐状态管理

```typescript
// YunYueBlog/components/MusicProvider.tsx
"use client";

import React, { createContext, useContext, useState, useRef, useEffect } from 'react';

interface Song {
  id: number;
  title: string;
  artist: string;
  cover: string;
  url: string;
  lyric: LyricLine[];
}

interface MusicContextType {
  playlist: Song[];
  currentSong: Song | null;
  isPlaying: boolean;
  currentTime: number;
  currentLyric: string;
  playMode: 'loop' | 'single' | 'random';
  togglePlay: () => void;
  nextSong: () => void;
  prevSong: () => void;
  setPlaylist: (songs: Song[]) => void;
}

const MusicContext = createContext<MusicContextType | null>(null);

export function MusicProvider({ children }: { children: React.ReactNode }) {
  const [playlist, setPlaylist] = useState<Song[]>([]);
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [playMode, setPlayMode] = useState<'loop' | 'single' | 'random'>('loop');
  const audioRef = useRef<HTMLAudioElement | null>(null);
  
  const currentSong = playlist[currentIndex] || null;
  const currentLyric = getCurrentLyric(currentSong?.lyric || [], currentTime);
  
  // 初始化音频
  useEffect(() => {
    audioRef.current = new Audio();
    audioRef.current.addEventListener('timeupdate', () => {
      setCurrentTime(audioRef.current?.currentTime || 0);
    });
    audioRef.current.addEventListener('ended', handleSongEnd);
    
    return () => {
      audioRef.current?.pause();
      audioRef.current?.removeEventListener('ended', handleSongEnd);
    };
  }, []);
  
  // 播放/暂停
  const togglePlay = () => {
    if (!audioRef.current || !currentSong) return;
    
    if (isPlaying) {
      audioRef.current.pause();
    } else {
      audioRef.current.src = currentSong.url;
      audioRef.current.play();
    }
    setIsPlaying(!isPlaying);
  };
  
  // 下一首
  const nextSong = () => {
    if (playlist.length === 0) return;
    
    let nextIndex: number;
    if (playMode === 'random') {
      nextIndex = Math.floor(Math.random() * playlist.length);
    } else if (playMode === 'single') {
      nextIndex = currentIndex;
    } else {
      nextIndex = (currentIndex + 1) % playlist.length;
    }
    
    setCurrentIndex(nextIndex);
    setIsPlaying(true);
  };
  
  // 歌曲结束处理
  const handleSongEnd = () => {
    if (playMode === 'single') {
      audioRef.current?.play();
    } else {
      nextSong();
    }
  };
  
  // 歌曲切换时播放
  useEffect(() => {
    if (audioRef.current && currentSong && isPlaying) {
      audioRef.current.src = currentSong.url;
      audioRef.current.play();
    }
  }, [currentIndex]);
  
  return (
    <MusicContext.Provider value={{
      playlist,
      currentSong,
      isPlaying,
      currentTime,
      currentLyric,
      playMode,
      togglePlay,
      nextSong,
      prevSong: () => setCurrentIndex((currentIndex - 1 + playlist.length) % playlist.length),
      setPlaylist,
    }}>
      {children}
    </MusicContext.Provider>
  );
}

// 获取当前歌词
function getCurrentLyric(lyrics: LyricLine[], time: number): string {
  for (let i = lyrics.length - 1; i >= 0; i--) {
    if (time >= lyrics[i].time) {
      return lyrics[i].text;
    }
  }
  return lyrics[0]?.text || '';
}

// 自定义 Hook
export function useMusic() {
  const context = useContext(MusicContext);
  if (!context) {
    throw new Error('useMusic must be used within MusicProvider');
  }
  return context;
}
```

```python
# my-blog-manager/cms_core/api/sync.py
import shutil
from pathlib import Path

class FileSyncManager:
    """文件同步管理器"""
    
    # 需要同步的文件夹
    SYNC_FOLDERS = ['posts', 'chatters', 'moments']
    
    # 需要同步的单文件
    SYNC_FILES = [
        'siteConfig.ts',
        'data/friends.ts',
        'data/projects.ts',
        'data/albums.ts',
    ]
    
    # 敏感字段（同步时过滤）
    SENSITIVE_KEYS = ['picBedToken', 'clientSecret']
    
    def __init__(self, source_dir: str, target_dir: str):
        self.source_dir = Path(source_dir)
        self.target_dir = Path(target_dir)
    
    def validate_target(self) -> bool:
        """验证目标目录是否为有效的博客目录"""
        package_json = self.target_dir / 'package.json'
        return package_json.exists()
    
    def sync_folders(self):
        """同步文件夹"""
        for folder in self.SYNC_FOLDERS:
            source = self.source_dir / folder
            target = self.target_dir / folder
            
            if not source.exists():
                continue
            
            # 清空目标文件夹
            if target.exists():
                shutil.rmtree(target)
            
            # 复制文件夹
            shutil.copytree(source, target)
            print(f"✅ 同步文件夹: {folder}")
    
    def sync_files(self):
        """同步单文件"""
        for file in self.SYNC_FILES:
            source = self.source_dir / file
            target = self.target_dir / file
            
            if not source.exists():
                continue
            
            # 确保目标目录存在
            target.parent.mkdir(parents=True, exist_ok=True)
            
            # 复制文件
            shutil.copy2(source, target)
            print(f"✅ 同步文件: {file}")
    
    def sync_site_config(self):
        """同步 siteConfig.ts（过滤敏感信息）"""
        source = self.source_dir / 'siteConfig.ts'
        target = self.target_dir / 'siteConfig.ts'
        
        if not source.exists():
            return
        
        # 读取源文件
        content = source.read_text(encoding='utf-8')
        
        # 过滤敏感字段
        for key in self.SENSITIVE_KEYS:
            # 使用正则匹配并移除敏感字段
            pattern = rf'{key}\s*:\s*["\'].*?["\']'
            content = re.sub(pattern, f'{key}: ""', content)
        
        # 写入目标文件
        target.write_text(content, encoding='utf-8')
        print("✅ 同步 siteConfig.ts（已过滤敏感信息）")
    
    def sync_all(self):
        """执行完整同步"""
        # 验证目标目录
        if not self.validate_target():
            raise ValueError("目标目录不是有效的博客目录")
        
        print("🔄 开始同步...")
        
        self.sync_folders()
        self.sync_files()
        self.sync_site_config()
        
        print("✅ 同步完成！")
```

### 3.2 配置文件同步

```python
# my-blog-manager/cms_core/api/config.py
import re
from typing import Any, Dict

class SiteConfigManager:
    """siteConfig.ts 配置管理器"""
    
    # 合法的根级字段白名单
    VALID_ROOT_KEYS = [
        'title', 'description', 'author', 'avatar',
        'social', 'gitalkConfig', 'geminiConfig', 'icpConfig',
        'footerBadges', 'buildDate', 'enableLevelSystem',
    ]
    
    def __init__(self, config_path: str):
        self.config_path = Path(config_path)
    
    def read(self) -> Dict[str, Any]:
        """读取配置文件"""
        content = self.config_path.read_text(encoding='utf-8')
        
        # 使用正则解析 TypeScript 对象
        config = {}
        
        # 解析简单键值对
        simple_pattern = r'(\w+)\s*:\s*["\']([^"\']*)["\']'
        for match in re.finditer(simple_pattern, content):
            key, value = match.groups()
            config[key] = value
        
        # 解析嵌套对象
        nested_pattern = r'(\w+)\s*:\s*\{([^}]+)\}'
        for match in re.finditer(nested_pattern, content):
            key, value_str = match.groups()
            config[key] = self._parse_nested(value_str)
        
        return config
    
    def write(self, key: str, value: Any):
        """写入配置"""
        # 验证字段合法性
        if key not in self.VALID_ROOT_KEYS:
            raise ValueError(f"非法字段: {key}")
        
        content = self.config_path.read_text(encoding='utf-8')
        
        # 序列化值
        serialized = self._serialize_value(value)
        
        # 查找并替换字段
        pattern = rf'({key}\s*:\s*)(.*?)(?=,\s*\w+\s*:|\s*\}})'
        replacement = rf'\g<1>{serialized}'
        
        new_content = re.sub(pattern, replacement, content, flags=re.DOTALL)
        
        self.config_path.write_text(new_content, encoding='utf-8')
    
    def _serialize_value(self, value: Any) -> str:
        """序列化值为 TypeScript 格式"""
        if isinstance(value, str):
            return f'"{value}"'
        elif isinstance(value, bool):
            return 'true' if value else 'false'
        elif isinstance(value, (int, float)):
            return str(value)
        elif isinstance(value, list):
            items = [self._serialize_value(item) for item in value]
            return f'[{", ".join(items)}]'
        elif isinstance(value, dict):
            items = [
                f'{k}: {self._serialize_value(v)}' 
                for k, v in value.items()
            ]
            return '{' + ', '.join(items) + '}'
        else:
            return str(value)
    
    def _parse_nested(self, value_str: str) -> Dict[str, Any]:
        """解析嵌套对象"""
        result = {}
        # 简化解析逻辑
        pattern = r'(\w+)\s*:\s*["\']([^"\']*)["\']'
        for match in re.finditer(pattern, value_str):
            key, value = match.groups()
            result[key] = value
        return result
```

```typescript
// YunYueBlog/app/api/chat/route.ts
import { NextResponse } from 'next/server';

// 支持 Gemini 和 OpenAI
const AI_PROVIDERS = {
  gemini: {
    url: 'https://generativelanguage.googleapis.com/v1beta/models/',
    buildRequest: (message: string, systemPrompt: string) => ({
      contents: [{ parts: [{ text: message }] }],
      systemInstruction: { parts: [{ text: systemPrompt }] },
      generationConfig: {
        maxOutputTokens: 150,
        temperature: 0.85,
      },
    }),
    extractReply: (data: any) => data.candidates?.[0]?.content?.parts?.[0]?.text,
  },
  openai: {
    url: 'https://api.openai.com/v1/chat/completions',
    buildRequest: (message: string, systemPrompt: string) => ({
      model: 'gpt-3.5-turbo',
      messages: [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: message },
      ],
      max_tokens: 150,
      temperature: 0.85,
    }),
    extractReply: (data: any) => data.choices?.[0]?.message?.content,
  },
};

export async function POST(request: Request) {
  try {
    const { message } = await request.json();
    
    // 获取配置
    const provider = process.env.AI_PROVIDER || 'gemini';
    const apiKey = process.env.GEMINI_API_KEY || process.env.OPENAI_API_KEY;
    const systemPrompt = process.env.AI_SYSTEM_PROMPT || 
      '你是一只叫"煤球"的傲娇暹罗猫，说话简短，句尾加"喵~"';
    
    if (!apiKey) {
      return NextResponse.json(
        { error: '未配置 AI API Key' },
        { status: 500 }
      );
    }
    
    const ai = AI_PROVIDERS[provider as keyof typeof AI_PROVIDERS];
    
    // 调用 AI API
    const response = await fetch(`${ai.url}${getModelId(provider)}:generateContent?key=${apiKey}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(ai.buildRequest(message, systemPrompt)),
    });
    
    if (!response.ok) {
      throw new Error(`AI API 错误: ${response.status}`);
    }
    
    const data = await response.json();
    const reply = ai.extractReply(data) || '本喵卡壳了喵...';
    
    return NextResponse.json({ success: true, reply });
  } catch (error) {
    console.error('AI 聊天错误:', error);
    return NextResponse.json(
      { error: 'AI 服务暂时不可用' },
      { status: 500 }
    );
  }
}

function getModelId(provider: string): string {
  switch (provider) {
    case 'gemini': return 'gemini-2.5-flash-lite';
    case 'openai': return 'gpt-3.5-turbo';
```

```python
from fastapi import APIRouter, UploadFile, File
import httpx

class PicBedManager:
    """图床管理器"""
    
    def __init__(self, url: str, token: str):
        self.url = url.rstrip('/')
        self.token = token
    
    async def test_token(self) -> dict:
        """验证 Token 有效性"""
        try:
            # 1. 检查服务是否在线
            async with httpx.AsyncClient() as client:
                response = await client.get(self.url, timeout=10)
                if response.status_code != 200:
                    return {"success": False, "message": "图床服务不在线"}
            
            # 2. 验证 Token 格式
            if not self.token.startswith('imgbed_'):
                return {"success": False, "message": "Token 格式错误"}
            
            # 3. 发送无文件上传请求验证 Token
            async with httpx.AsyncClient() as client:
                response = await client.post(
                    f"{self.url}/api/v1/upload",
                    headers={"Authorization": f"Bearer {self.token}"},
                    timeout=10
                )
                
                # 401 = Token 无效, 400 = Token 有效但缺少文件
                if response.status_code == 401:
                    return {"success": False, "message": "Token 无效"}
                elif response.status_code == 400:
                    return {"success": True, "message": "Token 有效"}
                else:
                    return {"success": True, "message": "验证通过"}
        
        except Exception as e:
            return {"success": False, "message": f"验证失败: {str(e)}"}
    
    async def upload_image(self, file: UploadFile) -> dict:
        """上传图片"""
        try:
            # 读取文件内容
            content = await file.read()
            
            # 上传到图床
            async with httpx.AsyncClient() as client:
                response = await client.post(
                    f"{self.url}/api/v1/upload",
                    headers={"Authorization": f"Bearer {self.token}"},
                    files={"file": (file.filename, content, file.content_type)},
                    timeout=30
                )
                
                if response.status_code != 200:
                    return {"success": False, "message": "上传失败"}
                
                data = response.json()
                
                # 提取直链
                image_url = data.get("data", {}).get("links", {}).get("url", "")
                
                return {
                    "success": True,
                    "url": image_url,
                    "message": "上传成功"
                }
        
        except Exception as e:
            return {"success": False, "message": f"上传失败: {str(e)}"}
```
